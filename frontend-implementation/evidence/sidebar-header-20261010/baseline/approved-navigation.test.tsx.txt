function deferred<T>() { let resolve!: (value: T) => void, reject!: (reason: unknown) => void;
  const promise = new Promise<T>((yes, no) => { resolve = yes; reject = no; }); return { promise, resolve, reject }; }

async function start() { render(<App />); fireEvent.click(projectAction("Open folder")); await screen.findByTitle("a.md"); }

const selected = () => document.querySelector('[role="tab"][aria-selected="true"]')?.textContent;
it("never displays a false unavailable alert while reopening released image pixels", async () => {
  await start(); await open("figure.png"); await screen.findByRole("img", { name: "figure.png" });
  await open("a.md");
  const pending = deferred<typeof pixels>(); mocks.openImage.mockReturnValueOnce(pending.promise);
  fireEvent.click(screen.getByRole("tab", { name: "figure.png" }));
  await waitFor(() => expect(mocks.openImage).toHaveBeenCalledTimes(2));
  expect(screen.queryByRole("alert")).toBeNull();
  expect(screen.getByText("Loading local image…")).toBeTruthy();
  await act(async () => pending.resolve(pixels)); await screen.findByRole("img", { name: "figure.png" });
  expect(screen.queryByRole("alert")).toBeNull();
});
it("shows a real image reactivation failure and keeps a later selected file stable", async () => {

});

it("a stale linked document response cannot apply its fragment to a newer file", async () => {

it("pinch zoom is file-specific, keeps chrome stable and leaves normal wheel scrolling alone", async () => {
  await start(); await open("a.md"); const reader = document.querySelector('.reader-scroll') ?? document.querySelector('[data-document-zoom]')?.parentElement;
  expect(reader).toBeTruthy();
  const ordinary = new WheelEvent('wheel', { deltaY: -60, bubbles: true, cancelable: true }); fireEvent(reader!, ordinary); expect(ordinary.defaultPrevented).toBe(false);
  const pinch = new WheelEvent('wheel', { deltaY: -100, ctrlKey: true, bubbles: true, cancelable: true, clientX: 100, clientY: 100 }); fireEvent(reader!, pinch);
  expect(pinch.defaultPrevented).toBe(true); expect(Number(document.querySelector('[data-document-zoom]')?.getAttribute('data-document-zoom'))).toBeGreaterThan(1);
  await open("b.md"); expect(document.querySelector('[data-document-zoom]')?.getAttribute('data-document-zoom')).toBe('1');
  fireEvent.click(screen.getByRole("tab", { name: "a.md" })); expect(Number(document.querySelector('[data-document-zoom]')?.getAttribute('data-document-zoom'))).toBeGreaterThan(1);

it("uses a dismissible narrow sidebar without overwriting the wider-window visibility choice", async () => {
  await start();
  fireEvent.click(projectsToggle()); // user closes the wide sidebar.
  const original = Object.getOwnPropertyDescriptor(window, "innerWidth")!;
  try {
    Object.defineProperty(window, "innerWidth", { configurable: true, value: 640 }); fireEvent(window, new Event("resize"));
    const sidebar = document.querySelector<HTMLElement>(".sidebar")!;
    expect(sidebar.dataset.compact).toBe("true"); expect(sidebar.dataset.open).toBe("false"); expect(sidebar.hasAttribute("inert")).toBe(true);
    fireEvent.click(projectsToggle());
    expect(sidebar.dataset.open).toBe("true"); expect(screen.getByRole("button", { name: "Close sidebar" })).toBeTruthy();
    Object.defineProperty(window, "innerWidth", { configurable: true, value: 1200 }); fireEvent(window, new Event("resize"));
    expect(sidebar.dataset.compact).toBe("false"); expect(sidebar.dataset.open).toBe("false");
  } finally { Object.defineProperty(window, "innerWidth", original); }
});
