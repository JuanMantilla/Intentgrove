14	    p.add_argument('executable', type=Path)

16	    qa = args.kind == 'qa'
17	    contents = b''.join(f.read_bytes() for f in args.assets.rglob('*') if f.is_file())
18	    found = [marker in contents for marker in MARKERS]
19	    if (qa and not all(found)) or (not qa and any(found)):
20	        p.error('frontend QA boundary mismatch')
21	    # Binary feature evidence is combined with cfg/build guards and runtime checks, not relied on alone.
22	    strings = subprocess.check_output(['strings', '-a', str(args.executable)])
23	    # Tauri's asset map retains filenames even when file bodies are compressed.
24	    for asset in args.assets.rglob('*'):
25	        if asset.is_file() and asset.suffix in ('.js', '.css') and asset.name.encode() not in strings:
26	            p.error('package does not embed the current asset set: '+asset.name)
27	    native = b'F002A_BOOTSTRAP_FD' in strings
28	    if native != qa:
29	        p.error('native QA boundary mismatch')
30	    if not qa and any(marker in strings for marker in MARKERS):
31	        p.error('QA frontend embedded in production')
32	    if not qa and any(marker in contents or marker in strings for marker in [b'diagram_acceptance', b'INTENTGROVE_DIAGRAM_ACCEPTANCE', b'INTENTGROVE_DIAG:', b'__diagMode']):
