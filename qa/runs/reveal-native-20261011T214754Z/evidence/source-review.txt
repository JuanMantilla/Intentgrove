19: fn root_path(state: &AppState, root_id: &str, segments: &[String]) -> AppResult<PathBuf> {
20:     if segments.is_empty() || segments.len() > 128 {
21:         return Err(AppError::new("invalid_path", "No local file selected"));
22:     }
23:     for segment in segments {
24:         validate_segment(segment)?;
25:     }
26:     let root = state.root(root_id)?;
27:     let scoped = open_relative(&root.file, segments, false)?;
28:     let path = segments
29:         .iter()
30:         .fold(root.path, |path, part| path.join(part));
31:     // Confirm the path handed to the desktop still names the scoped file.
32:     if identity(&open_selected(&path, false)?)? != identity(&scoped)? {
33:         return Err(AppError::new(
34:             "reselection_required",
35:             "Selected file changed",
36:         ));
37:     }

93: fn reveal_native(path: &std::path::Path) -> AppResult<()> {
94:     use std::ffi::c_void;
95:     use std::os::windows::ffi::OsStrExt;
96:     #[link(name = "shell32")]
97:     extern "system" {
98:         fn SHParseDisplayName(
99:             name: *const u16,
100:             context: *const c_void,
101:             pidl: *mut *mut c_void,
102:             attributes: u32,
103:             result: *mut u32,
104:         ) -> i32;
105:         fn SHOpenFolderAndSelectItems(
106:             pidl: *const c_void,
107:             count: u32,
108:             children: *const *const c_void,
109:             flags: u32,
110:         ) -> i32;
111:     }
112:     #[link(name = "ole32")]
113:     extern "system" {
114:         fn CoInitializeEx(reserved: *const c_void, flags: u32) -> i32;
115:         fn CoUninitialize();
116:         fn CoTaskMemFree(memory: *mut c_void);
117:     }
118:     fn failure(code: i32) -> AppError {
119:         AppError::new(
120:             "reveal_failed",
121:             format!("File Explorer HRESULT 0x{:08X}", code as u32),
122:         )
123:     }
124:     let name: Vec<u16> = path.as_os_str().encode_wide().chain(Some(0)).collect();
125:     unsafe {
126:         let initialized = CoInitializeEx(std::ptr::null(), 2);
127:         // A Tauri pool thread may already belong to a different COM apartment.
128:         if initialized < 0 && initialized != 0x80010106u32 as i32 {
129:             return Err(failure(initialized));
130:         }
131:         let mut pidl = std::ptr::null_mut();
132:         let parsed = SHParseDisplayName(
133:             name.as_ptr(),
134:             std::ptr::null(),
135:             &mut pidl,
136:             0,
137:             std::ptr::null_mut(),
138:         );
139:         let result = if parsed < 0 {
140:             Err(failure(parsed))
141:         } else if pidl.is_null() {
142:             Err(AppError::new("reveal_failed", "No local Shell item"))
143:         } else {
144:             let opened = SHOpenFolderAndSelectItems(pidl, 0, std::ptr::null(), 0);
145:             if opened < 0 {
146:                 Err(failure(opened))
147:             } else {
148:                 Ok(())
149:             }
150:         };
151:         if !pidl.is_null() {
152:             CoTaskMemFree(pidl);
153:         }
154:         if initialized >= 0 {
155:             CoUninitialize();
156:         }
157:         result
158:     }

7: // Preserve the framework delegate and its other methods. Only its termination
8: // decision is wrapped; Cancel keeps the native app alive for the editing prompt.
9: bool md_install_quit_guard_on_delegate(id delegate, MDQuitGuard guard) {
10:     if (!delegate || !guard) return false;
11:     Class delegateClass = object_getClass(delegate);
12:     SEL selector = @selector(applicationShouldTerminate:);
13:     Method original = class_getInstanceMethod(delegateClass, selector);
14:     IMP previous = original ? method_getImplementation(original) : NULL;
15:     const char *encoding = original ? method_getTypeEncoding(original) : "q@:@";
16:     IMP replacement = imp_implementationWithBlock(^NSApplicationTerminateReply(id receiver, NSApplication *sender) {
17:         if (guard()) return NSTerminateCancel;
18:         if (previous) {
19:             typedef NSApplicationTerminateReply (*Decision)(id, SEL, NSApplication *);
20:             return ((Decision)previous)(receiver, selector, sender);
21:         }
22:         return NSTerminateNow;
23:     });
24:     class_replaceMethod(delegateClass, selector, replacement, encoding);
25:     return method_getImplementation(class_getInstanceMethod(delegateClass, selector)) == replacement;
26: }
27: 
28: bool md_install_quit_guard(MDQuitGuard guard) {
29:     if (![NSThread isMainThread]) return false;
30:     return md_install_quit_guard_on_delegate([NSApp delegate], guard);
